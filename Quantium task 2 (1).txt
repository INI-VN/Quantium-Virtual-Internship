##Create a function to calculate correlation for a measure, looping through each control store
##let Input_table be metric table with potential comparison stores
##let Metrics be measure for calculating correlation
##let Trial_store be the store to which input stores will be compared. 

Cal_corr <- function(Input_table, Metrics, Trial_store) {
  calc_corr_table = data.table(store1 = numeric(), store2 = numeric(), corr_measure = numeric())
  store_numbers <- unique(Input_table[, STORE_NBR])
  for (i in store_numbers) {
      cal_measure = data.table("store1" = Trial_store, 
                                     "store2" = i, 
                                     "corr_measure" = cor(Input_table[STORE_NBR == 
                                                                        Trial_store, eval(Metrics)],
                                                          Input_table[STORE_NBR == i,
                                                                      eval(Metrics)])
      )                            
      calc_corr_table <- rbind(calc_corr_table, cal_measure)}
  return(calc_corr_table)
}


##Create a function to calculate correlation for a measure, looping through each control store
Cal_mag_distance <- function(Input_table, Metrics, Trial_store) {
  calc_dist_table = data.table(store1 = numeric(), store2 = numeric(), YEARMONTH = 
                               numeric(), measure = numeric())
  store_numbers <- unique(Input_table[, STORE_NBR])
  
  for (i in store_numbers) {
    cal_measure = data.table("store1" = Trial_store, 
                            "store2" = i,
                            "YEARMONTH" = Input_table[STORE_NBR == Trial_store, YEARMONTH],
                            "measure" = abs(Input_table[STORE_NBR == Trial_store, eval(Metrics)]
                                                     - Input_table[STORE_NBR == i,eval(Metrics)])
    )
    calc_dist_table <- rbind(calc_dist_table, cal_measure)
  }
  min_max_dist = calc_dist_table[, .(min_dist = min(measure), max_dist = max(measure)), 
                             by = c("store1", "YEARMONTH")]
  dist_table = merge(calc_dist_table, min_max_dist, by = c("store1", "YEARMONTH"))
  dist_table[, magnitude_measure := 1 - (measure - min_dist)/(max_dist - min_dist)]
  
  final_dist_table <- dist_table[, .(mag_measure = mean(magnitude_measure)), by = 
                                .(store1, store2)]
  return(final_dist_table)
}
